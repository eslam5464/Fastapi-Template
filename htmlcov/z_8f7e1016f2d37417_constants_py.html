1class RateLimitPrefix: 

2 """ 

3 Centralized registry of all rate limit key prefixes. 

4 

5 All rate limit keys follow the pattern: ratelimit:{category}:{identifier} 

6 where identifier is typically an IP address or user ID. 

7 

8 Example: 

9 ```python 

10 from app.core.constants import RateLimitPrefix 

11 

12 # Use in rate limiting 

13 key = f"{RateLimitPrefix.AUTH}{ip_address}" 

14 # Result: "ratelimit:auth:192.168.1.1" 

15 ``` 

16 """

36 @classmethod 

37 def all_prefixes(cls) -> set[str]: 

38 """ 

39 Get all registered prefixes for validation. 

40 

41 Returns: 

42 set[str]: Set of all registered rate limit prefixes 

43 

44 Example: 

45 ```python 

46 prefixes = RateLimitPrefix.all_prefixes() 

47 # {'ratelimit:auth:', 'ratelimit:user:', ...} 

48 ``` 

49 """ 

50 return { 

51 value 

52 for key, value in cls.__dict__.items() 

53 if isinstance(value, str) and value.startswith("ratelimit:") 

54 }

56 @classmethod 

57 def validate_prefix(cls, prefix: str) -> None: 

58 """ 

59 Validate that a prefix doesn't conflict with existing ones. 

60 

61 Args: 

62 prefix: The prefix to validate (should include "ratelimit:" and trailing ":") 

63 

64 Raises: 

65 ValueError: If prefix already exists in the registry 

66 

67 Example: 

68 ```python 

69 # This will raise ValueError if prefix already exists 

70 RateLimitPrefix.validate_prefix("ratelimit:custom:") 

71 ``` 

72 """ 

73 if prefix in cls.all_prefixes(): 

74 raise ValueError( 

75 f"Rate limit prefix '{prefix}' is already registered. " 

76 f"Existing prefixes: {cls.all_prefixes()}" 

77 )
